import os
from typing import Protocol
from dotenv import load_dotenv
from src.core.logger import logger

load_dotenv()

class AffiliateHandler(Protocol):
    def apply_affiliate_link(self, url: str) -> str:
        ...

class AmazonAffiliateHandler:
    def __init__(self):
        self.tag = os.getenv("AMAZON_ID", "default_tag")

    def apply_affiliate_link(self, url: str) -> str:
        if "amazon" in url.lower():
            separator = "&" if "?" in url else "?"
            return f"{url}{separator}tag={self.tag}"
        return url

class AliExpressAffiliateHandler:
    def __init__(self):
        self.tag = os.getenv("ALIEXPRESS_ID", "default_tag")

    def apply_affiliate_link(self, url: str) -> str:
        if "aliexpress" in url.lower():
            separator = "&" if "?" in url else "?"
            return f"{url}{separator}aff_id={self.tag}"
        return url

class GenericAffiliateHandler:
    def __init__(self):
        self.tag = os.getenv("GENERIC_ID", "default_tag")

    def apply_affiliate_link(self, url: str) -> str:
        separator = "&" if "?" in url else "?"
        return f"{url}{separator}ref={self.tag}"

class AffiliateService:
    """
    Orchestrates affiliate link transformation based on the store.
    """
    def __init__(self):
        self.handlers = {
            "amazon": AmazonAffiliateHandler(),
            "aliexpress": AliExpressAffiliateHandler(),
            "generic": GenericAffiliateHandler()
        }

    def transform(self, url: str, store_name: str) -> str:
        store_key = store_name.lower()
        handler = self.handlers.get(store_key, self.handlers["generic"])
        return handler.apply_affiliate_link(url)

import logging
import sys

def setup_logger() -> logging.Logger:
    """
    Configures and returns the central logger for the application.
    """
    logger = logging.getLogger("EcoOfertasBot")
    logger.setLevel(logging.INFO)

    # Avoid duplicate handlers if setup_logger is called multiple times
    if not logger.handlers:
        formatter = logging.Formatter(
            fmt="%(asctime)s [%(levelname)s] %(name)s: %(message)s",
            datefmt="%Y-%m-%d %H:%M:%S"
        )

        # Console handler
        console_handler = logging.StreamHandler(sys.stdout)
        console_handler.setFormatter(formatter)
        logger.addHandler(console_handler)

    return logger

# Global logger instance
logger = setup_logger()

import json
import os
from typing import List, Dict, Any
from src.core.models import Oferta
from src.core.logger import logger

class StorageManager:
    """
    Handles persistence of seen offers to prevent duplicates.
    """
    def __init__(self, db_path: str = "data/ofertas.json"):
        self.db_path = db_path
        self._ensure_db_exists()

    def _ensure_db_exists(self):
        if not os.path.exists(self.db_path):
            os.makedirs(os.path.dirname(self.db_path), exist_ok=True)
            self.save_db({})

    def load_db(self) -> Dict[str, Any]:
        try:
            with open(self.db_path, 'r', encoding='utf-8') as f:
                return json.load(f)
        except (json.JSONDecodeError, IOError) as e:
            logger.error(f"Error loading database {self.db_path}: {e}")
            return {}

    def save_db(self, data: Dict[str, Any]):
        try:
            with open(self.db_path, 'w', encoding='utf-8') as f:
                json.dump(data, f, indent=4, ensure_ascii=False)
        except IOError as e:
            logger.error(f"Error saving database {self.db_path}: {e}")

    def is_new(self, oferta: Oferta) -> bool:
        db = self.load_db()
        return oferta.title not in db

    def mark_as_seen(self, oferta: Oferta):
        db = self.load_db()
        db[oferta.title] = {
            "price": oferta.price,
            "last_seen": None # Could be extended with timestamps
        }
        self.save_db(db)

import requests
from bs4 import BeautifulSoup
from typing import List, Optional
from src.core.models import Oferta, StoreConfig
from src.core.logger import logger

class BaseScraper:
    """
    Provides common utilities for all scrapers.
    """
    def __init__(self):
        self.headers = {
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.124 Safari/537.36"
        }

    def get_soup(self, url: str) -> Optional[BeautifulSoup]:
        try:
            response = requests.get(url, headers=self.headers, timeout=10)
            response.raise_for_status()
            return BeautifulSoup(response.text, 'html.parser')
        except requests.RequestException as e:
            logger.error(f"Request failed for {url}: {e}")
            return None

    def clean_price(self, price_text: str) -> float:
        """
        Extracts numeric value from price strings (e.g., '$12.99' -> 12.99).
        """
        cleaned = "".join(c for c in price_text if c.isdigit() or c == '.')
        try:
            return float(cleaned) if cleaned else 0.0
        except ValueError:
            logger.warning(f"Could not parse price text: {price_text}")
            return 0.0

from typing import List
from bs4 import BeautifulSoup
from src.core.models import Oferta, StoreConfig
from src.scrapers.base import BaseScraper
from src.core.logger import logger

class GenericScraper(BaseScraper):
    """
    A configuration-driven scraper that can work with any store
    that follows a simple HTML structure.
    """
    def scrape(self, config: StoreConfig) -> List[Oferta]:
        logger.info(f"Scanning store: {config.name} at {config.base_url}")
        soup = self.get_soup(config.base_url)

        if not soup:
            return []

        offers = []
        # Find all product containers based on the selector
        products = soup.select(config.product_selector)

        for p in products:
            try:
                # Extract title
                title_el = p.select_one(config.title_selector)
                title = title_el.get_text(strip=True) if title_el else "Unknown Product"

                # Extract price
                price_el = p.select_one(config.price_selector)
                price_text = price_el.get_text(strip=True) if price_el else "0"
                price = self.clean_price(price_text)

                # Extract URL (handle relative paths)
                link_el = p.select_one(config.title_selector if not config.image_selector else config.title_selector)
                # Default to title selector for URL if not specified
                url = link_el['href'] if link_el and link_el.has_attr('href') else ""
                if url and not url.startswith('http'):
                    # Simple relative URL resolution
                    url = config.base_url + url if config.base_url.endswith('/') else config.base_url + '/' + url

                # Extract Image
                image_url = None
                if config.image_selector:
                    img_el = p.select_one(config.image_selector)
                    if img_el and img_el.has_attr('src'):
                        image_url = img_el['src']
                        if image_url and not image_url.startswith('http'):
                            image_url = config.base_url + image_url if config.base_url.endswith('/') else config.base_url + '/' + image_url

                offers.append(Oferta(
                    title=title,
                    price=price,
                    url=url,
                    category="General", # Default, can be expanded
                    store=config.name,
                    image_url=image_url
                ))
            except Exception as e:
                logger.warning(f"Failed to parse a product in {config.name}: {e}")
                continue

        logger.info(f"Successfully extracted {len(offers)} offers from {config.name}")
        return offers

import os
from typing import List
from src.core.logger import logger
from src.core.models import Oferta, StoreConfig
from src.core.storage import StorageManager
from src.scrapers.generic import GenericScraper
from src.affiliates.providers import AffiliateService
from src.web.generator import WebGenerator

def main():
    logger.info("ðŸš€ Starting EcoOfertasBot Orchestrator...")

    # 1. Configuration
    # In a real scenario, this could be loaded from a config.json file
    stores_to_scan = [
        StoreConfig(
            name="BooksToScrape",
            base_url="https://books.toscrape.com/",
            product_selector="article.product_pod",
            title_selector="h3 a",
            price_selector=".price_color",
            image_selector="img"
        ),
        # Add more stores here as needed
    ]

    # 2. Initialize Components
    storage = StorageManager()
    scraper = GenericScraper()
    affiliate_service = AffiliateService()
    web_gen = WebGenerator()

    all_new_offers: List[Oferta] = []

    # 3. Execution Loop
    for config in stores_to_scan:
        try:
            # Step A: Scrape products
            raw_offers = scraper.scrape(config)

            # Step B: Filter new ones and apply affiliate links
            for offer in raw_offers:
                if storage.is_new(offer):
                    # Transform URL to affiliate link
                    affiliate_url = affiliate_service.transform(offer.url, config.name)

                    # Create a new Oferta object with the affiliate link
                    final_offer = Oferta(
                        title=offer.title,
                        price=offer.price,
                        url=affiliate_url,
                        category="General", # Can be customized per store
                        store=config.name,
                        image_url=offer.image_url
                    )

                    all_new_offers.append(final_offer)
                    storage.mark_as_seen(final_offer)

        except Exception as e:
            logger.error(f"Critical error processing {config.name}: {e}")

    # 4. Finalization
    if all_new_offers:
        logger.info(f"Found {len(all_new_offers)} new offers! Updating website...")
        # We pass all offers from the DB to the web generator to keep the site full
        # For this version, we'll just use the new ones or reload all from DB
        db_data = storage.load_db()
        all_offers_for_web = [
            Oferta(
                title=title,
                price=val["price"],
                url="#", # In a full version, we'd store the affiliate URL in the DB too
                category="General",
                store="Various"
            )
            for title, val in db_data.items()
        ]
        # Add the newly found ones
        all_offers_for_web.extend(all_new_offers)

        web_gen.generate(all_offers_for_web)
        logger.info("âœ¨ Cycle completed successfully.")
    else:
        logger.info("No new offers found in this cycle.")

if __name__ == "__main__":
    main()
